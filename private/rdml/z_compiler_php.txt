function stmt(x)
{
	if (inSubr == "Y")  				// cache the subroutine body, output later
	{
		subroutine_body += x;
		subroutine_body += '\n';
		return '';
	}
	if (inSubr == "N")					// cache the main body, output later
	{
		main_body += x;
		main_body += '\n';
		return '';
	}
	if (inSubr == "F")					// just found endroutine - send subroutine body
	{
		inSubr = "N";
		return x + '\n';				// return subroutine body
	}
}

function findFieldType(f,t)
{
	
	// look in FILE definitions
	for(i=0;i<tables.length;i=i+2) 
	{
		var file  = tables[i];
		var tfields = tables[i+1];
		if (file == t)
		{
			for(i=0;i<tfields.length;i=i+3) 
			{
				var name = tfields[i];
				var type = tfields[i+1];
				var key  = tfields[i+2];
				if (name == f)
				{
					bindType = getMysqliBindType(type);
					found = true;
					break;  // leave fields loop
				}
			}
			break; // leave files loop
		}
	}
	
	if (found == true)
	{
		bindType += "T";  // indicates a table field so this should be a variable
		return bindType;
	}

	// find string definition in local DEFINES (if not found above)
	
	var i = 0;
	var bindType = ''; 
	var found = false;
	for(i=0;i<fields_stack.length;i=i+2) {
		var fld  = fields_stack[i];
		var type = fields_stack[i+1];
		if (fld == f)
		{
			if (type == "*CHAR")
			{
				bindType = 's';
			}
			if (type == "*DEC")
			{
				bindType = 'd';
			}
			found = true;
			break;
		}
	}
	
	if (found == true)
	{
		bindType += "L"; 		// indicates a local field so substitution variable
		return bindType;
	}

	// not found so assume string and local : this is a repository field and not implemented yet
	bindType = 'sL'; 
	return bindType;
}
function getMysqliBindType(mysqlType) {
  // Normalize: uppercase, strip size/precision, remove modifiers
  var type = mysqlType.trim().toUpperCase();
  type = type.replace(/\(.*\)/, '');
  type = type.replace(/\s+(UNSIGNED|ZEROFILL|SIGNED)/g, '');
  type = type.trim();

  var intTypes = [
    'TINYINT', 'SMALLINT', 'MEDIUMINT', 'INT', 'INTEGER',
    'BIT', 'BOOL', 'BOOLEAN', 'YEAR'
  ];

  var floatTypes = [
    'FLOAT', 'DOUBLE', 'DOUBLE PRECISION', 'REAL', 'DECIMAL'
  ];

  var stringTypes = [
    'CHAR', 'VARCHAR',
    'TINYTEXT', 'TEXT', 'MEDIUMTEXT', 'LONGTEXT',
    'DECIMAL', 'DEC', 'NUMERIC', 'FIXED',
    'BIGINT',
    'DATE', 'TIME', 'DATETIME', 'TIMESTAMP',
    'ENUM', 'SET', 'JSON',
    'GEOMETRY', 'POINT', 'LINESTRING', 'POLYGON',
    'MULTIPOINT', 'MULTILINESTRING', 'MULTIPOLYGON',
    'GEOMETRYCOLLECTION'
  ];

  var blobTypes = [
    'TINYBLOB', 'BLOB', 'MEDIUMBLOB', 'LONGBLOB',
    'BINARY', 'VARBINARY'
  ];

  if (intTypes.indexOf(type) !== -1) {
    return 'i';
  }
  if (floatTypes.indexOf(type) !== -1) {
    return 'd';
  }
  if (stringTypes.indexOf(type) !== -1) {
    return 's';
  }
  if (blobTypes.indexOf(type) !== -1) {
    return 'b';
  }

  print('Unknown MySQL type: ' + type + " - defaulting to 's'");
  return 's';
}

function formatWhere(w,t)
{

	var fullWhere = '';
	var fullType = '';
	var fullBind = '';

	function formatOne(token)
	{

		debug("token=[" + token + "]");

		if (/^[()]$/.test(token)) {								// paranthesis
			return token;
		
		} else if (/^(OR|AND|=|<|<=|>|>=)$/i.test(token)) {		// operators
			return token;

		} else if (/'([^']*)'/.test(token)) {					// single quoted literal			
			return '"' + token.slice(1,-1) + '"';

		} else if (/"[^"]*"/.test(token)) {						// double quoted literal				
			return token;

		} else if (/^-?\d+(\.\d+)?$/.test(token)) {				// number literal
			return token;
	
    	} else if (/#([a-zA-Z0-9_]+)/.test(token)) {			// variable
			var field = token.slice(1).toUpperCase();
			var bnd  = findFieldType(field,t);
			var bindType = bnd[0];		
			var bindContext = bnd[1];
			if (bindContext == 'T')
			{
				return field;
			}
			if (bindContext == 'L')
			{
				fullType += bindType;
				fullBind += ',$' + field;
				return "?";
			}	
		} else {												// naked literals leftover
			return '"' + token + '"';
		}
	}

	// extract each token (field, paranthesis, logical operator, literal, or quoted literal)
	fullWhere = w.replace(/"[^"]*"|'[^']*'|\(|\)|-?\d+(\.\d+)?|[^\s]+/g, formatOne);
	return [fullWhere,fullType,fullBind]; 

}

function endroutine()
{
	inSubr = "F";												// this is global - set the finish flag
	
	var myY = subroutine_func_def;								// output function definition
	
	myY += "global $IO_STS; \n";
	myY += "global $IO_ROWS; \n";
	myY += "global $IO_ERR; \n";
	
	for (var i=0; i<subroutine_id_stack.length; i++) {			// output global variable name definitions (to put all variable in subroutine in global namespace)
		myY += "global $" + subroutine_id_stack[i] + " ; \n";
	}
	
	for (var i=0; i<subroutine_lists.length; i++) {				// output global list name definitions (to put all lists in subroutine in global namespace)
		myY += "global $" + subroutine_lists[i] + " ; \n";
		myY += "global $_" + subroutine_lists[i] + "_ptr ; \n";
	}
	
	myY += 'list(';												// set the parms passed into function from the parm list
	for (var i=0; i<subroutine_return.length; i++) {			
		myY += subroutine_return[i];
		if (i < subroutine_return.length - 1) myY += ',';
	}
	myY += ')=$_sub_parms; \n';
	
	myY += subroutine_body;										// output the cached subroutine body
	
	myY += 'return array(';										// output the cached return variables
	for (var i=0; i<subroutine_return.length; i++) {			
		myY += subroutine_return[i];
		if (i < subroutine_return.length - 1) myY += ',';
	}
	myY += '); \n';
																	
	myY += '}';
	subroutine_id_stack = new Array();							// clear id global array 
	subroutine_return   = new Array();							// clear return parm array
	subroutine_body = '';										// clear the cached subroutine body
	subroutine_lists = new Array();								// clear the cached def_list names
	return myY;
}

function L_emit_field(x)
{
	var myY  = "";
	
	// new section...cache fields in symbol table for init routine at end
	// myY+= "LIST_TO_L_EMIT_FIELDS:" + x.toSource() + "\n";
	for(i=0;i<x.length;i++) {
	    var fname = x[i];
		variablestack.push(fname);   // is global
	}
	
	// remove lists from variable stack - we will output those later
	var local_def_list = new Array();							
	var array_name    = '';										
	for(i=0;i<def_list_stack.length;i++) {						// spin the DEF_LIST array
		local_def_list = def_list_stack[i];						// get next DEF_LIST off array
		array_name = local_def_list[0];							// get the LIST name
		for (var j=0; j<variablestack.length; j++) {
			fldName = variablestack[j][0];
			if (array_name == fldName) {
				variablestack.splice(j,1);						// remove DEF_LIST name from variablestack
				def_list_screen.push(local_def_list);			// copy DEF_LIST definition to screen list array
				break;									
			}
		}
	}

	// Build the Tiny Bit Strong Template Calls
	myY += "$TBS = new \\clsTinyButStrong;\n";
	myY += "$TBS->SetOption('noerr', false);\n";
	myY += "$TBS->LoadTemplate($CurrentTemplate);\n";
	
	// Build each list call
	var local_def_list = new Array();							
	var array_name    = '';										
	for(i=0;i<def_list_screen.length;i++) {						// spin the DEF_LIST array
		local_def_list = def_list_screen[i];					// get next DEF_LIST off array
		array_name = local_def_list[0];							// get the LIST name
		myY += "$TBS->MergeBlock('" + array_name + "',$" + array_name + ");\n";
	}
	
	myY += "$TBS->Show();\n";
	myY += "exit(); \n"; 
	
	
	return myY;
}
